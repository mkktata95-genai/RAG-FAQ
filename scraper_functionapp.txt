import subprocess, sys, re, tempfile, csv
FEED = "https://artefacts.royallondongroup.com/pypi/rlg-pypi/simple"
rows = []
for line in open(sys.argv[1]):
    line = line.strip()
    if not line or line.startswith("#"):
        continue
    name, ver = line.split("==")
    with tempfile.TemporaryDirectory() as d:
        p = subprocess.run([sys.executable, "-m", "pip", "download", line, "--no-deps", "-d", d,
                            "--only-binary=:all:", "--platform", "manylinux2014_x86_64",
                            "--python-version", "3.11", "--index-url", FEED],
                           capture_output=True, text=True)
    out = p.stdout + p.stderr
    if p.returncode == 0:
        rows.append((name, ver, "OK", ""))
    else:
        m = re.search(r"from versions: ([^)]*)\)", out)
        rows.append((name, ver, "MISSING" if m else "ERROR",
                     (m.group(1) if m else out.strip().splitlines()[-1])[:200]))
for r in rows:
    print(*r, sep=" | ")
csv.writer(open("feedcheck.csv", "w", newline="")).writerows(
    [("package", "ours", "status", "feed_versions_or_error"), *rows])
